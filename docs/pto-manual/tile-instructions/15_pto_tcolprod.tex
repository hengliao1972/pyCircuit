\subsection{\texttt{TCOLPROD}}
\noindent\textbf{Bundle encoding.} UOP entry 15, opcode \texttt{0x101A}. Bundle: \texttt{B.OP + B.ARG + B.DIM + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x101A}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src}, \texttt{SrcTile1}=\texttt{absent}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size. \texttt{B.DIM} supplies column-axis reduction or column-broadcast bounds.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Reduce each column by multiplying across rows.
\paragraph{Example.}
Source 4x3 tile. Output[0,2] = src[0,2] * src[1,2] * src[2,2] * src[3,2].
\par\smallskip
\paragraph{PTO ISA operation.}
Let \texttt{R = src.GetValidRow()} and \texttt{C = src.GetValidCol()}. For \texttt{0 \textless{}= j \textless{} C}:

\[
\mathrm{dst}_{0,j} = \prod_{i=0}^{R-1} \mathrm{src}_{i,j}
\]
}\par\smallskip
\paragraph{Notes.}
For FP types, the product of zero values is zero. An odd number of -1 values yields -1. NaN propagates.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_tcolprod(src, dst):
    dtype = dst.element_type
    valid_rows, valid_cols = src.valid_shape

    lanes = pto.get_lanes(dtype)
    remained = valid_cols

    for col_chunk in range(0, valid_cols, lanes):
        mask, remained = pto.make_mask(dtype, remained)

        acc = pto.vlds(src[0, col_chunk:])
        for row in range(1, valid_rows, 1):
            row_vec = pto.vlds(src[row, col_chunk:])
            acc = pto.vmul(acc, row_vec, mask)
        pto.vsts(acc, dst[0, col_chunk:], mask)

    return
\end{lstlisting}
