\subsection{\texttt{TMIN}}
\noindent\textbf{Bundle encoding.} UOP entry 28, opcode \texttt{0x1036}. Bundle: \texttt{B.OP + B.ARG + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1036}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src0}, \texttt{SrcTile1}=\texttt{src1}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Elementwise minimum of two tiles.
\paragraph{Example.}
[1, 4, 2] min [3, 2, 5] = [1, 2, 2].
\par\smallskip
\paragraph{PTO ISA operation.}
For each element \texttt{(i, j)} in the valid region:

\[
\mathrm{dst}_{i,j} = \min(\mathrm{src0}_{i,j}, \mathrm{src1}_{i,j})
\]
}\par\smallskip
\paragraph{Notes.}
Same tie-breaking and NaN rules as \texttt{TMAX}.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_tmin(src0, src1, dst):
    dtype = dst.element_type
    valid_rows, valid_cols = dst.valid_shape

    for row in range(0, valid_rows, 1):
        remained = valid_cols
        for col in range(0, valid_cols, pto.get_lanes(dtype)):
            mask, remained = pto.make_mask(dtype, remained)
            lhs = pto.vlds(src0[row, col:])
            rhs = pto.vlds(src1[row, col:])
            min_val = pto.vmin(lhs, rhs, mask)
            pto.vsts(min_val, dst[row, col:], mask)
    return
\end{lstlisting}
