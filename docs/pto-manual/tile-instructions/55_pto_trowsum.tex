\subsection{\texttt{TROWSUM}}
\noindent\textbf{Bundle encoding.} UOP entry 55, opcode \texttt{0x1054}. Bundle: \texttt{B.OP + B.ARG + B.DIM + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1054}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src}, \texttt{SrcTile1}=\texttt{absent}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size. \texttt{B.DIM} supplies row-axis reduction or row-broadcast bounds. Scratch operand(s) \texttt{tmp} are implementation-local and are not encoded in the CPU-visible bundle.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Reduce each row by summing across columns.
\paragraph{Example.}
Source row [1, 2, 3]: output[0] = 6.
\par\smallskip
\paragraph{PTO ISA operation.}
Let \texttt{R = src.GetValidRow()} and \texttt{C = src.GetValidCol()}. For \texttt{0 \textless{}= i \textless{} R}:

\[
\mathrm{dst}_{i,0} = \sum_{j=0}^{C-1} \mathrm{src}_{i,j}
\]
}\par\smallskip
\paragraph{Notes.}
Integer sum wraps modulo 2\textsuperscript{64}. FP sum follows IEEE 754 with VARIANT-encoded rounding mode.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_trowsum(src, tmp, dst):
    src_dtype = src.element_type
    dst_dtype = dst.element_type

    if pto.constexpr(src_dtype == pto.i16):
        acc_dtype = pto.i32
    else:
        acc_dtype = src_dtype

    lanes = pto.get_lanes(src_dtype)
    valid_rows, valid_cols = src.valid_shape

    zero_val = acc_dtype(0)

    elem_bytes = pto.bytewidth(dst_dtype)
    if pto.constexpr(elem_bytes == 4):
        store_dist = pto.VStoreDist.ONE_POINT_B32
    elif pto.constexpr(elem_bytes == 2):
        store_dist = pto.VStoreDist.ONE_POINT_B16
    else:
        store_dist = pto.VStoreDist.ONE_POINT_B8

    dst_mask_1, _ = pto.make_mask(dst_dtype, 1)

    for row in range(0, valid_rows, 1):
        remained = valid_cols

        acc_mask_1, _ = pto.make_mask(acc_dtype, 1)

        v_acc = pto.vbr(zero_val)

        for col in range(0, valid_cols, lanes):
            mask, remained = pto.make_mask(src_dtype, remained)
            v_src = pto.vlds(src[row, col:])

            v_reduced = pto.vcadd(v_src, mask)

            v_acc = pto.vadd(v_acc, v_reduced, acc_mask_1)

        if pto.constexpr(acc_dtype != dst_dtype):
            acc_mask_for_cvt, _ = pto.make_mask(acc_dtype, 1)
            v_acc_casted = pto.vcvt(v_acc, dst_dtype, acc_mask_for_cvt, sat=pto.VcvtSatMode.SAT, part=pto.VcvtPartMode.EVEN)
            pto.vsts(v_acc_casted, dst[row, 0:], dst_mask_1, dist=store_dist)
        else:
            pto.vsts(v_acc, dst[row, 0:], dst_mask_1, dist=store_dist)
    return
\end{lstlisting}
