\subsection{\texttt{TRELU}}
\noindent\textbf{Bundle encoding.} UOP entry 41, opcode \texttt{0x1049}. Bundle: \texttt{B.OP + B.ARG + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1049}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src}, \texttt{SrcTile1}=\texttt{absent}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Elementwise ReLU of a tile.
\paragraph{Example.}
Tile elements [-2, -1, 0, 1, 2] give [0, 0, 0, 1, 2].
\par\smallskip
\paragraph{PTO ISA operation.}
For each element \texttt{(i, j)} in the valid region:

\[
\mathrm{dst}_{i,j} = \max(\mathrm{src}_{i,j}, 0)
\]
}\par\smallskip
\paragraph{Notes.}
Zero is exactly representable in IEEE 754 and is produced for inputs in (-0, 0]. The negative zero sign is preserved: -0.0 maps to +0.0.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_trelu(src, dst):
    dtype = dst.element_type
    valid_rows, valid_cols = dst.valid_shape

    lanes = pto.get_lanes(dtype)

    for row in range(0, valid_rows, 1):
        remained = valid_cols
        for col in range(0, valid_cols, lanes):
            mask, remained = pto.make_mask(dtype, remained)
            src_vec = pto.vlds(src[row, col:])
            result = pto.vrelu(src_vec, mask)
            pto.vsts(result, dst[row, col:], mask)
    return
\end{lstlisting}
