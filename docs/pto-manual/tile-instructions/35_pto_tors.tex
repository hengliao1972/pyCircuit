\subsection{\texttt{TORS}}
\noindent\textbf{Bundle encoding.} UOP entry 35, opcode \texttt{0x1040}. Bundle: \texttt{B.OP + B.ARG + B.IOR + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1040}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src}, \texttt{SrcTile1}=\texttt{absent}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size. \texttt{B.IOR} imports scalar operand(s): \texttt{scalar}.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Elementwise bitwise OR of a tile and a scalar.
\paragraph{Example.}
Tile [0b00001010] OR 0xF0 = [0b11111010].
\par\smallskip
\paragraph{PTO ISA operation.}
For each element \texttt{(i, j)} in the valid region:

\[
\mathrm{dst}_{i,j} = \mathrm{src}_{i,j} \;|\; \mathrm{scalar}
\]
}\par\smallskip
\paragraph{Notes.}
The scalar is imported via \texttt{B.IOR}.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_tors(src, scalar, dst):
    dtype = src.element_type
    valid_rows, valid_cols = src.valid_shape

    for row in range(0, valid_rows, 1):
        remained = valid_cols
        for col in range(0, valid_cols, pto.get_lanes(dtype)):
            mask, remained = pto.make_mask(dtype, remained)
            vec = pto.vlds(src[row, col:])
            scalar_vec = pto.vbr(scalar)
            result = pto.vor(vec, scalar_vec, mask)
            pto.vsts(result, dst[row, col:], mask)
    return
\end{lstlisting}
