\subsection{\texttt{TEXPANDS}}
\noindent\textbf{Bundle encoding.} UOP entry 21, opcode \texttt{0x1020}. Bundle: \texttt{B.OP + B.ARG + B.IOR + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1020}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{absent}, \texttt{SrcTile1}=\texttt{absent}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size. \texttt{B.IOR} imports scalar operand(s): \texttt{scalar}.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Broadcast a scalar into a destination tile.
\paragraph{Example.}
Scalar 0 broadcast to a 4x4 tile produces a tile of all zeros.
\par\smallskip
\paragraph{PTO ISA operation.}
For each element \texttt{(i, j)} in the valid region:

\[
\mathrm{dst}_{i,j} = \mathrm{scalar}
\]
}\par\smallskip
\paragraph{Notes.}
The scalar is imported via \texttt{B.IOR}. The destination tile's shape and format are defined by the \texttt{B.IOTI} destination selector; the scalar is cast to the destination element type.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_texpands(scalar, dst):
    dtype = dst.element_type
    valid_rows, valid_cols = dst.valid_shape

    for row in range(0, valid_rows, 1):
        remained = valid_cols
        for col in range(0, valid_cols, pto.get_lanes(dtype)):
            mask, remained = pto.make_mask(dtype, remained)
            vec = pto.vdup(scalar, mask)
            pto.vsts(vec, dst[row, col:], mask)

    return
\end{lstlisting}
