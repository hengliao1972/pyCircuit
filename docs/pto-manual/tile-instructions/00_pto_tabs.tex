\subsection{\texttt{TABS}}
\noindent\textbf{Bundle encoding.} UOP entry 0, opcode \texttt{0x1006}. Bundle: \texttt{B.OP + B.ARG + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1006}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src}, \texttt{SrcTile1}=\texttt{absent}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Elementwise absolute value of a tile.
\paragraph{Example.}
For a 2x3 tile [[-3, 2, -7], [0, -1, 8]], the result is [[3, 2, 7], [0, 1, 8]].
\par\smallskip
\paragraph{PTO ISA operation.}
For each element \texttt{(i, j)} in the valid region:

\[
\mathrm{dst}_{i,j} = \left|\mathrm{src}_{i,j}\right|
\]
}\par\smallskip
\paragraph{Notes.}
Unsigned integer inputs (\texttt{ui8}, \texttt{ui16}, \texttt{ui32}) are unchanged. Signed integer overflow wraps: \texttt{INT\_MIN} maps to \texttt{INT\_MIN} (abs is undefined at the minimum).

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_tabs(src, dst):
    dtype = dst.element_type
    valid_rows, valid_cols = dst.valid_shape

    for row in range(0, valid_rows, 1):
        remained = valid_cols
        for col in range(0, valid_cols, pto.get_lanes(dtype)):
            mask, remained = pto.make_mask(dtype, remained)
            vinput = pto.vlds(src[row, col:])
            result = pto.vabs(vinput, mask)
            pto.vsts(result, dst[row, col:], mask)
    return
\end{lstlisting}
