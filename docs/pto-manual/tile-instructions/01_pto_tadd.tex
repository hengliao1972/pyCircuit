\subsection{\texttt{TADD}}
\noindent\textbf{Bundle encoding.} UOP entry 1, opcode \texttt{0x1007}. Bundle: \texttt{B.OP + B.IOTI}. \texttt{B.OP.Opcode}=\texttt{0x1007}. \texttt{B.IOTI}: selector=\texttt{111 last}, \texttt{SrcTile0}=\texttt{src0}, \texttt{SrcTile1}=\texttt{src1}, \texttt{DstTile}=\texttt{dst}, \texttt{SizeCode}=profile tile size.\par\smallskip
{\small
\paragraph{PTO ISA description.}
Elementwise add of two tiles.
\paragraph{Example.}
tile\_a + tile\_b: element at (i,j) of the result equals element at (i,j) of tile\_a plus element at (i,j) of tile\_b.
\par\smallskip
\paragraph{PTO ISA operation.}
For each element \texttt{(i, j)} in the valid region:

\[
\mathrm{dst}_{i,j} = \mathrm{src0}_{i,j} + \mathrm{src1}_{i,j}
\]
}\par\smallskip
\paragraph{Notes.}
All three tiles MUST have the same valid rows and valid columns. FP addition follows IEEE 754 with the rounding mode encoded in the VARIANT field.

\paragraph{Microcode body DSL.}
\begin{lstlisting}[style=ptocode]
def uop_tadd(src0, src1, dst):
    dtype = dst.element_type
    valid_rows, valid_cols = dst.valid_shape

    for row in range(0, valid_rows, 1):
        remained = valid_cols
        for col in range(0, valid_cols, pto.get_lanes(dtype)):
            mask, remained = pto.make_mask(dtype, remained)
            lhs = pto.vlds(src0[row, col:])
            rhs = pto.vlds(src1[row, col:])
            summed = pto.vadd(lhs, rhs, mask)
            pto.vsts(summed, dst[row, col:], mask)
    return
\end{lstlisting}
