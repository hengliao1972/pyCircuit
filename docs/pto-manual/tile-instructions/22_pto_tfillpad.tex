\section{\texttt{TFILLPAD}}
\noindent\textbf{Descriptor profile.} Vec entry 22, tile catalog opcode \texttt{0x1023}. Descriptor bundle: \texttt{B.OP + B.ATTR + B.IOT}; WO/WM sites: 8; VEC beat-level sites: 8; ROM bytes: 64.\par\smallskip
\TileInstructionEncoding{figs/encoding/tile_instruction_entries/entry_22_pto_tfillpad_th0.pdf}{figs/encoding/tile_instruction_entries/entry_22_pto_tfillpad_th2.pdf}{0x1023}{22}{B.OP + B.ATTR + B.IOT}
\paragraph{Bundled descriptor (PTO-ISA header).}
\begin{description}[leftmargin=0.18\linewidth,style=nextline]
\item[Bundle] \texttt{B.OP + B.ATTR + B.IOT}. \texttt{B.OP.desc\_count} = 2; \texttt{B.OP.rom\_entry\_id} = 22.
\item[Assembly] 0: \texttt{B.OP} selects \texttt{TFILLPAD}, carries element format/variant, declares \texttt{desc\_count}, and names the ROM entry. 1: \texttt{B.ATTR} carries pad/fill and valid-region policy. 2: \texttt{B.IOT} binds architectural tile operands. Across all \texttt{B.IOT} descriptors: tile inputs \texttt{src}, mask inputs none, and outputs \texttt{dst}.
\item[Tile operands] 1 tile input(s): \texttt{src}; 1 tile output(s): \texttt{dst}. Bound by 1 architectural \texttt{B.IOT} descriptor; each \texttt{B.IOT} supplies at most two source selectors plus one destination selector.
\item[Scalar GPR operands] No scalar GPR import/export descriptor is required.
\item[Metadata operands] \texttt{B.ATTR} carries pad/fill and valid-region policy.
\end{description}
\par\smallskip
{\small
\paragraph{PTO ISA description.}
Copy a source tile into a destination tile and fill the remaining (padded) elements with a compile-time pad value selected by \texttt{TileDataDst::PadVal} (e.g., \texttt{PadValue::Min}/\texttt{PadValue::Max}).

This is commonly used to materialize deterministic values outside the runtime valid region so that subsequent ops can operate on a full static tile shape.
\paragraph{PTO ISA operation.}
Let \texttt{VR = src.GetValidRow()} and \texttt{VC = src.GetValidCol()}. For each destination element \texttt{(i, j)}:

\[
\mathrm{dst}_{i,j} =
\begin{cases}
\mathrm{src}_{i,j} & \text{if } i < VR \text{ and } j < VC \\
\mathrm{pad}       & \text{otherwise}
\end{cases}
\]

\texttt{pad} is determined by \texttt{TileDataDst::PadVal} and the element type (e.g., \texttt{+inf/-inf} for floating types when available, otherwise \texttt{std::numeric\_limits\textless{}T\textgreater{}::max()/min()}).
}\par\smallskip

\paragraph{ROM body DSL.}
\begin{lstlisting}[style=ptocode]
# Supported (src dtype, dst dtype) pairs:
# (f32, f32), (bf16, bf16), (i16, i16), (si16, si16),
# (ui16, ui16), (i32, i32), (si32, si32), (ui32, ui32),
# (i8, i8), (si8, si8), (ui8, ui8)

def rom_tfillpad(src, dst):
    dtype = src.element_type
    src_valid_rows, src_valid_cols = src.valid_shape
    dst_valid_rows, dst_valid_cols = dst.valid_shape
    dst_rows, dst_cols = dst.shape

    lanes = pto.get_lanes(dtype)
    aligned_col = (src_valid_cols // lanes) * lanes
    has_tail = (src_valid_cols % lanes) != 0
    has_valid_expansion = (src_valid_rows < dst_valid_rows) or (src_valid_cols < dst_valid_cols)

    if pto.constexpr(dtype == pto.f32):
        if pto.constexpr(dst.pad_value == pto.PadValue.ZERO and has_valid_expansion):
            fill_scalar = pto.f32(_NEG1_F32)
        elif pto.constexpr(dst.pad_value == pto.PadValue.ZERO):
            fill_scalar = pto.f32(0.0)
        elif pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.f32(0.0)
    elif pto.constexpr(dtype == pto.bf16):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.bf16(0)
    elif pto.constexpr(dtype == pto.ui16):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.ui16(0)
    elif pto.constexpr(dtype == pto.si16):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.si16(0)
    elif pto.constexpr(dtype == pto.i16):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.i16(0)
    elif pto.constexpr(dtype == pto.ui32):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.ui32(0)
    elif pto.constexpr(dtype == pto.si32):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.si32(0)
    elif pto.constexpr(dtype == pto.i32):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.i32(0)
    elif pto.constexpr(dtype == pto.ui8):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.ui8(0)
    elif pto.constexpr(dtype == pto.si8):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.si8(0)
    elif pto.constexpr(dtype == pto.i8):
        if pto.constexpr(dst.pad_value != pto.PadValue.NULL):
            fill_scalar = dst.pad_value.eval(dtype)
        else:
            fill_scalar = pto.i8(0)

    for row in range(0, src_valid_rows, 1):
        remained = src_valid_cols
        for col in range(0, aligned_col, lanes):
            mask, remained = pto.make_mask(dtype, remained)
            data = pto.vlds(src[row, col:])
            pto.vsts(data, dst[row, col:], mask)

    if pto.constexpr(aligned_col < dst_cols):
        for row in range(0, src_valid_rows, 1):
            for col in range(aligned_col, dst_cols, lanes):
                mask, remained = pto.make_mask(dtype, remained)
                vec = pto.vdup(fill_scalar, mask)
                pto.vsts(vec, dst[row, col:], mask)

    if pto.constexpr(has_tail):
        for row in range(0, src_valid_rows, 1):
            mask_copy, remained = pto.make_mask(dtype, src_valid_cols - aligned_col)
            data = pto.vlds(src[row, aligned_col:])
            pto.vsts(data, dst[row, aligned_col:], mask_copy)

    if pto.constexpr(src_valid_rows < dst_rows):
        for row in range(src_valid_rows, dst_rows, 1):
            for col in range(0, dst_cols, lanes):
                mask, remained = pto.make_mask(dtype, remained)
                vec = pto.vdup(fill_scalar, mask)
                pto.vsts(vec, dst[row, col:], mask)
    return
\end{lstlisting}
